@section Scripts {
    <script src="https://cdnjs.cloudflare.com/ajax/libs/xlsx/0.18.5/xlsx.full.min.js"></script>
    <script>
        (function() {
            'use strict';
            var tokenEl = document.querySelector('input[name="__RequestVerificationToken"]');
            var csrfToken = tokenEl ? tokenEl.value : '';

            function ajaxHeaders() {
                return { 'RequestVerificationToken': csrfToken };
            }

            var classes = @Html.Raw(Json.Serialize(ViewBag.Classes));

            $(function() {
                var sel = document.getElementById('classFilter');
                if (classes && classes.length) {
                    classes.forEach(function(c) {
                        var o = document.createElement('option');
                        o.value = c.id; o.textContent = c.name;
                        sel.appendChild(o);
                    });
                }
            });

            function getRequest() {
                return {
                    dateFrom: document.getElementById('dateFrom').value || null,
                    dateTo: document.getElementById('dateTo').value || null,
                    classId: parseInt(document.getElementById('classFilter').value) || null,
                    status: parseInt(document.getElementById('statusFilter').value) || null,
                    gender: document.getElementById('genderFilter').value || null,
                    religion: document.getElementById('religionFilter').value || null
                };
            }

            function loadRegister() {
                var container = document.getElementById('registerContainer');
                $.ajax({
                    url: '/Admission/GetRegisterReport', type: 'POST', contentType: 'application/json',
                    headers: ajaxHeaders(),
                    data: JSON.stringify(getRequest()),
                    success: function(res) {
                        if (!res.success) {
                            container.innerHTML = '<div class="adm-alert adm-alert--error">' + res.message + '</div>';
                            return;
                        }
                        var data = res.data;
                        if (!data.rows || !data.rows.length) {
                            container.innerHTML = '<div class="adm-card"><div class="adm-card__body" style="text-align:center;padding:40px;"><p>No records found for the selected filters.</p></div></div>';
                            return;
                        }
                        var html = '<div class="adm-card"><div class="adm-card__header"><h3>Admission Register</h3><span>' + data.totalRecords + ' records</span></div><div class="adm-card__body"><div class="adm-table-wrap"><table class="adm-table adm-table--striped"><thead><tr><th>#</th><th>App No</th><th>Name</th><th>DOB</th><th>Gender</th><th>Father</th><th>Mother</th><th>Mobile</th><th>Religion</th><th>Class</th><th>Status</th><th>Submitted</th></tr></thead><tbody>';
                        data.rows.forEach(function(r) {
                            var dob = r.dateOfBirth ? new Date(r.dateOfBirth).toLocaleDateString() : '';
                            var sub = r.submittedAt ? new Date(r.submittedAt).toLocaleDateString() : '';
                            html += '<tr><td>' + r.serialNo + '</td><td>' + escHtml(r.applicationNo) + '</td><td>' + escHtml(r.applicantName) + '</td><td>' + dob + '</td><td>' + escHtml(r.gender) + '</td><td>' + escHtml(r.fatherName) + '</td><td>' + escHtml(r.motherName) + '</td><td>' + escHtml(r.mobile) + '</td><td>' + escHtml(r.religion) + '</td><td>' + escHtml(r.appliedClass) + '</td><td><span class="adm-badge adm-badge--' + statusClass(r.status) + '">' + escHtml(r.status) + '</span></td><td>' + sub + '</td></tr>';
                        });
                        html += '</tbody></table></div></div></div>';
                        container.innerHTML = html;
                    },
                    error: function() { container.innerHTML = '<div class="adm-alert adm-alert--error">Failed to load report.</div>'; }
                });
            }

            function statusClass(s) {
                switch ((s || '').toLowerCase()) {
                    case 'pending': return 'warning';
                    case 'approved': return 'info';
                    case 'rejected': return 'error';
                    case 'converted': return 'success';
                    default: return 'neutral';
                }
            }

            function escHtml(str) {
                if (!str) return '—';
                var d = document.createElement('div');
                d.textContent = str;
                return d.innerHTML;
            }

            function exportExcel() {
                $.ajax({
                    url: '/Admission/ExportReportExcel', type: 'POST', contentType: 'application/json',
                    headers: ajaxHeaders(),
                    data: JSON.stringify(getRequest()),
                    xhrFields: { responseType: 'blob' },
                    success: function(blob) {
                        var link = document.createElement('a');
                        link.href = URL.createObjectURL(blob);
                        link.download = 'admission_register_' + Date.now() + '.xlsx';
                        link.click();
                    },
                    error: function() { alert('Export failed.'); }
                });
            }

            window.loadRegister = loadRegister;
            window.exportExcel = exportExcel;
        })();
    </script>
}
